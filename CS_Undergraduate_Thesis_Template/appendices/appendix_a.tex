import os
import re
import json
import time
import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin

class LawphilStatutoryScraper:
    """Autonomous scraper for national statutes from the Lawphil repository."""
    BASE_URL = "https://lawphil.net/"
    USER_AGENT = "AcademicResearch-LegalNLP-AdDU/1.0"

    def __init__(self, output_file="data/national_statutes.jsonl"):
        self.output_file = output_file
        self.session = requests.Session()
        self.session.headers.update({"User-Agent": self.USER_AGENT})

    def fetch_statute(self, relative_url, statute_type, statute_num):
        """Fetches and parses a single statute HTML document."""
        full_url = urljoin(self.BASE_URL, relative_url)
        response = self.session.get(full_url, timeout=15)
        response.raise_for_status()
        
        soup = BeautifulSoup(response.content, "html.parser")
        main_content = soup.find("td", {"valign": "top"}) or soup.find("div", {"id": "content"})
        raw_text = main_content.get_text(separator="\n") if main_content else soup.get_text()
        
        return self.parse_provisions(raw_text, statute_type, statute_num, full_url)

    def parse_provisions(self, text, statute_type, statute_num, source_url):
        """Extracts substantive sections while maintaining structural hierarchy."""
        lines = [line.strip() for line in text.split("\n") if line.strip()]
        title = lines[0] if lines else f"{statute_type} {statute_num}"
        
        # Section boundary regex pattern
        sec_pattern = re.compile(r'^(Section|Sec\.)\s+(\d+[a-zA-Z]?)\.?\s*(.*)', re.IGNORECASE)
        provisions = []
        current_sec = None
        current_body = []

        for line in lines:
            match = sec_pattern.match(line)
            if match:
                if current_sec and current_body:
                    provisions.append({
                        "section_id": f"{statute_type}_{statute_num}_S{current_sec}",
                        "section_num": current_sec,
                        "statutory_text": " ".join(current_body).strip()
                    })
                current_sec = match.group(2)
                current_body = [match.group(3)] if match.group(3) else []
            elif current_sec:
                current_body.append(line)

        if current_sec and current_body:
            provisions.append({
                "section_id": f"{statute_type}_{statute_num}_S{current_sec}",
                "section_num": current_sec,
                "statutory_text": " ".join(current_body).strip()
            })

        return {
            "statute_id": f"{statute_type}_{statute_num}",
            "statute_title": title,
            "source_url": source_url,
            "provisions": provisions
        }

    def serialize_jsonl(self, records):
        """Streams unpolluted statutory records to JSON Lines format."""
        with open(self.output_file, "a", encoding="utf-8") as f:
            for rec in records:
                f.write(json.dumps(rec, ensure_ascii=False) + "\n")
