\section{Statutory Scraper and Boundary Parser}
\label{sec:app_code_scraping}

\begin{lstlisting}[language=Python, caption={Lawphil Statutory Scraper and Provision Boundary Serializer}, label={lst:code_scraper_lawphil}]
import os
import re
import json
import time
import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin

class LawphilStatutoryScraper:
    """Autonomous scraper for national statutes from the Lawphil repository."""
    BASE_URL = "https://lawphil.net/"
    USER_AGENT = "AcademicResearch-LegalNLP-AdDU/1.0"

    def __init__(self, output_dir="data/raw_statutes"):
        self.output_dir = output_dir
        self.session = requests.Session()
        self.session.headers.update({"User-Agent": self.USER_AGENT})
        os.makedirs(output_dir, exist_ok=True)

    def extract_provisions(self, text, statute_type, statute_num):
        """Extracts substantive sections while maintaining structural hierarchy."""
        lines = [line.strip() for line in text.split("\n") if line.strip()]
        title = lines[0] if lines else f"{statute_type} {statute_num}"
        
        # Section boundary regex pattern
        sec_pattern = re.compile(r'^(Section|Sec\.)\s+(\d+[a-zA-Z]?)\.?\s*(.*)', re.IGNORECASE)
        provisions = []
        current_sec = None
        current_body = []
        
        for line in lines:
            match = sec_pattern.match(line)
            if match:
                if current_sec and current_body:
                    provisions.append({
                        "section_id": f"{statute_type}_{statute_num}_S{current_sec}",
                        "section_num": current_sec,
                        "text": " ".join(current_body)
                    })
                current_sec = match.group(2)
                current_body = [match.group(3)] if match.group(3) else []
            elif current_sec:
                current_body.append(line)
                
        # Append final boundary
        if current_sec and current_body:
            provisions.append({
                "section_id": f"{statute_type}_{statute_num}_S{current_sec}",
                "section_num": current_sec,
                "text": " ".join(current_body)
            })
        return provisions
\end{lstlisting}
