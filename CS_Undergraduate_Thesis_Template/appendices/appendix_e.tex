\section{Cross-Encoder Inference Engine}
\label{sec:app_code_cross_encoder}

\begin{lstlisting}[language=Python, caption={Stage 2 Cross-Encoder Inference and Calibrated Triage}, label={lst:code_stage2_inference}]
import torch
import torch.nn.functional as F
from typing import List, Dict, Tuple
from transformers import AutoTokenizer, AutoModelForSequenceClassification

class LegalConflictInferenceEngine:
    """Stage 2 fine cross-encoder inference with calibrated decision thresholds."""

    def __init__(self, model_path: str = "models/deberta-v3-legal-nli", threshold: float = 0.45):
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.tokenizer = AutoTokenizer.from_pretrained(model_path)
        self.model = AutoModelForSequenceClassification.from_pretrained(model_path)
        self.model.to(self.device)
        self.model.eval()
        self.threshold = threshold  # Calibrated tau* = 0.45 (C(FN)/C(FP) = 5.0)

    def classify_pair(self, statute_premise: str, ordinance_hypothesis: str) -> Dict:
        """Evaluates an asymmetric premise-hypothesis pair and returns calibrated triage."""
        inputs = self.tokenizer(
            statute_premise,
            ordinance_hypothesis,
            truncation=True,
            max_length=512,
            padding=True,
            return_tensors="pt"
        ).to(self.device)

        with torch.no_grad():
            outputs = self.model(**inputs)
            probs = F.softmax(outputs.logits, dim=-1).squeeze().cpu().numpy()

        # Class indexing: 0: Entailment, 1: Neutral, 2: Contradiction
        p_entail, p_neutral, p_contra = float(probs[0]), float(probs[1]), float(probs[2])
        
        # Apply cost-sensitive binary triage boundary
        is_conflict = p_contra >= self.threshold
        
        predicted_label = "Contradiction" if is_conflict else ("Entailment" if p_entail > p_neutral else "Neutral")

        return {
            "predicted_label": predicted_label,
            "conflict_risk": p_contra,
            "entailment_prob": p_entail,
            "neutral_prob": p_neutral,
            "triage_flag": is_conflict,
            "calibrated_threshold": self.threshold
        }

    def batch_audit(self, pairs: List[Tuple[str, str]], batch_size: int = 16) -> List[Dict]:
        """Executes batched cross-encoder inference using dynamic token padding."""
        results = []
        for i in range(0, len(pairs), batch_size):
            batch = pairs[i:i + batch_size]
            premises = [p[0] for p in batch]
            hypotheses = [p[1] for p in batch]

            inputs = self.tokenizer(
                premises,
                hypotheses,
                truncation=True,
                max_length=512,
                padding=True,  # Dynamic padding to longest sequence in batch
                return_tensors="pt"
            ).to(self.device)

            with torch.no_grad():
                logits = self.model(**inputs).logits
                probs = F.softmax(logits, dim=-1).cpu().numpy()

            for prob in probs:
                p_contra = float(prob[2])
                results.append({
                    "conflict_risk": p_contra,
                    "triage_flag": p_contra >= self.threshold
                })
        return results
